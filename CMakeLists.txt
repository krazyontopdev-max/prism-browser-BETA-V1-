cmake_minimum_required(VERSION 3.21)
project(Prism VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(PRISM_ENABLE_SCREEN_SHARING "Enable screen sharing with a validated WebEngine build" OFF)

find_package(Qt6 6.12 REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel Concurrent)

if(WIN32)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/imageformats")
    configure_file(third_party/qt-webp/plugins/imageformats/qwebp.dll
        "${CMAKE_BINARY_DIR}/imageformats/qwebp.dll" COPYONLY)
endif()

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/wallpapers/catalog.json")
file(COPY wallpapers/catalog.json wallpapers/images wallpapers/CREDITS.html
    wallpapers/licenses wallpapers/sources.json DESTINATION "${CMAKE_BINARY_DIR}/wallpapers")
file(COPY third_party/filters/ DESTINATION "${CMAKE_BINARY_DIR}/filters")

add_library(prism_core STATIC src/address.cpp src/address.h src/browserwindow.cpp src/browserwindow.h
    src/newtabstore.cpp src/newtabstore.h src/newtabbridge.cpp src/newtabbridge.h
    src/shields.cpp src/shields.h src/browserdata.cpp src/browserdata.h src/webdata.cpp src/webdata.h
    src/localfilepicker.cpp src/localfilepicker.h src/settings.cpp)
target_include_directories(prism_core PUBLIC src)
target_compile_definitions(prism_core PRIVATE PRISM_ENABLE_SCREEN_SHARING=$<BOOL:${PRISM_ENABLE_SCREEN_SHARING}>)
target_link_libraries(prism_core PUBLIC Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel Qt6::Concurrent)
if(MSVC)
    target_compile_options(prism_core PUBLIC /W4 /permissive- /utf-8)
endif()

qt_add_executable(Prism WIN32 src/main.cpp resources/prism.qrc)
target_link_libraries(Prism PRIVATE prism_core)
target_compile_definitions(Prism PRIVATE PRISM_VERSION="${PROJECT_VERSION}")
if(WIN32)
    enable_language(RC)
    configure_file(resources/prism-version.rc.in "${CMAKE_BINARY_DIR}/prism-version.rc" @ONLY)
    target_sources(Prism PRIVATE "${CMAKE_BINARY_DIR}/prism-version.rc")
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(prism_capture_fixture WIN32 tests/capture_fixture.cpp)
    target_link_libraries(prism_capture_fixture PRIVATE Qt6::Widgets)
    add_executable(prism_media_probe tests/media_probe.cpp)
    target_link_libraries(prism_media_probe PRIVATE prism_core Qt6::Test)
    add_executable(prism_privacy_tests tests/privacy_tests.cpp)
    target_link_libraries(prism_privacy_tests PRIVATE prism_core Qt6::Test)
    add_test(NAME prism_privacy_tests COMMAND prism_privacy_tests -o "${CMAKE_BINARY_DIR}/prism_privacy_tests-results.txt,txt")
    add_executable(prism_tests tests/browser_tests.cpp)
    target_link_libraries(prism_tests PRIVATE prism_core Qt6::Test)
    add_test(NAME prism_tests COMMAND prism_tests -o "${CMAKE_BINARY_DIR}/prism_tests-results.txt,txt")
    set_tests_properties(prism_tests PROPERTIES TIMEOUT 120)
    add_executable(prism_window_tests tests/window_tests.cpp resources/prism.qrc)
    add_dependencies(prism_window_tests prism_capture_fixture)
    target_link_libraries(prism_window_tests PRIVATE prism_core Qt6::Test)
    add_test(NAME prism_window_tests COMMAND prism_window_tests -o "${CMAKE_BINARY_DIR}/prism_window_tests-results.txt,txt")
    set_tests_properties(prism_window_tests PROPERTIES TIMEOUT 180)
    add_executable(prism_newtab_tests tests/newtab_tests.cpp resources/prism.qrc)
    target_link_libraries(prism_newtab_tests PRIVATE prism_core Qt6::Test)
    add_test(NAME prism_newtab_tests COMMAND prism_newtab_tests -o "${CMAKE_BINARY_DIR}/prism_newtab_tests-results.txt,txt")
    set_tests_properties(prism_newtab_tests PROPERTIES TIMEOUT 90)
    add_executable(prism_benchmark tests/performance.cpp resources/prism.qrc)
    target_link_libraries(prism_benchmark PRIVATE prism_core Qt6::Test)
    if(WIN32)
        target_link_libraries(prism_benchmark PRIVATE psapi)
    endif()
endif()
